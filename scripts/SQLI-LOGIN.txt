DB_PATH    = os.path.join(os.path.dirname(__file__), 'db.sqlite3')

APP_TITLE  = os.environ.get('APP_TITLE',  'CorpNet Portal')

def load_flag_into_db():
    if not os.path.exists(FLAG_PATH):
        return
    with open(FLAG_PATH) as f:
        flag = f.read().strip()
    conn = sqlite3.connect(DB_PATH)
    conn.execute("UPDATE secrets SET secret = ? WHERE owner = 'admin'", (flag,))
    conn.commit()
    conn.close()

@app.before_request
def _reload_flag():
    # The worker writes the flag file during the PLACE_FLAGS stage, which runs
    # AFTER this app was started in EXECUTE_SCRIPTS. Loading the flag only at
    # import time would therefore always miss it (the dashboard/UNION extraction
    # would return the 'flag_not_placed_yet' placeholder). Reloading per request
    # ensures the real flag is loaded into the secrets table once it is placed.
    load_flag_into_db()

@app.route('/', methods=['GET', 'POST'])
def login():
    error = None
    if request.method == 'POST':
        username = request.form.get('username', '')
        password = request.form.get('password', '')
        try:
            conn = sqlite3.connect(DB_PATH)
            c = conn.cursor()
            # VULNERABLE: password field concatenated directly into the query
            query = f"SELECT id, username, role FROM users WHERE username = ? AND password = '{password}'"
            c.execute(query, (username,))
            row = c.fetchone()
            if row:
                session['user_id'] = row[0]
                session['username'] = row[1]
                session['role'] = row[2]
                return redirect('/dashboard')
            else:
                error = 'Invalid credentials'
        except Exception as e:
            error = f'DB error: {e}' if DIFFICULTY == 'easy' else 'Invalid credentials'
        finally:
            conn.close()
    return render_template_string(LOGIN_HTML, error=error, app_title=APP_TITLE)

@app.route('/dashboard')
def dashboard():
    if 'username' not in session:
        return redirect('/')
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    c.execute("SELECT secret FROM secrets WHERE owner = ?", (session['username'],))
    row = c.fetchone()
    secret = row[0] if row else None
    conn.close()
    return render_template_string(DASHBOARD_HTML,
        username=session['username'],
        role=session['role'],
        secret=secret,
        app_title=APP_TITLE)

@app.route('/logout')
def logout():
    session.clear()
    return redirect('/')

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=int(os.environ.get('PORT', 5000)))
